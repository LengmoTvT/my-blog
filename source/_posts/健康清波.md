于是我们需要它的 三个接口分别是：
# 接口
url_get_numlist = "http://jkscapi.zwjk.com//jkscapi/Hospital/GetNumList"    # 前置查询号源接口
url_save_user = "http://jkscapi.zwjk.com//jkscapi/JZRUser/Save"             # 接口1 添加用户
url_save_yy = "http://jkscapi.zwjk.com//jkscapi/Hospital/SaveYY"            # 接口2 预约提交
